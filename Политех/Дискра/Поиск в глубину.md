### Основа DFS

1. Выбираем текущую вершину (входящая, помеченная) $v$.
2. Рассмотрим её окрестность $N(v)$. Ищем непомеченные вершины в окрестности — выбираем любую и переходим в неё. Т.е. помечаем и продолжаем.
3. Если нет непомеченных вершин — возврат назад к родительской вершине, из которой была помечена текущая. И данная вершина стала непросмотренной. Если возвращаться некуда — выход.

Пусть в графе $G$ проведён поиск в глубину. Обозначим через $T$ множество всех древесных рёбер (по которым шли в алгоритме). Все остальные рёбра графа будем называть <u>обратными рёбрами</u>. Множество всех обратных рёбер будем обозначать через $B$.

Изложим версию алгоритма поиска в глубину, основанную на рекурсивной процедуре $DFS(v)$, осуществляющей поиск в глубину из вершины $v$. Название алгоритма является аббревиатурой от англ. *depth first search*.

```
procedure DFS(v);
begin
  num[v] := i; i := i+1;
  for u ∈ list[v] do
    if num[u] = 0 then
      begin
        T := T ∪ {uv}; father[u] := v; DFS(v)
      end
    else if num[u] < num[v] and u ≠ father[v] then
      B := B ∪ {uv}
end;

begin
  i := 1; T := ∅; B := ∅;
  for v ∈ V do num[v] := 0;
  for v ∈ V do
    if num[v] = 0 then
    begin
      father[v] := ∅; DFS(v)
    end
end.
```

Тот же алгоритм на Си — структура, имена (`num`, `father`, `list`, `T`, `B`) и порядок действий повторяют псевдокод один в один:

```c
#include <stdio.h>

#define MAXV 101               /* вершины нумеруем с 1 */
#define MAXE 10001             /* рёбер не больше, чем вершин*вершин */

int n, m;                       /* n вершин, m рёбер */
int list[MAXV][MAXV];           /* list[v] — список смежности вершины v */
int listSize[MAXV];

int num[MAXV];                  /* num[v] = 0 — вершина ещё не помечена */
int father[MAXV];               /* father[v] — родитель; 0 играет роль ∅ */
int i;                          /* счётчик пометок i из псевдокода */

typedef struct { int u, v; } Edge;
Edge T[MAXE], B[MAXE];          /* множества древесных и обратных рёбер */
int tCount = 0, bCount = 0;

/* procedure DFS(v); */
void dfs(int v)
{
    num[v] = i; i = i + 1;                 /* num[v] := i;  i := i+1 */

    for (int k = 0; k < listSize[v]; k++) {/* for u ∈ list[v] do */
        int u = list[v][k];
        if (num[u] == 0) {                 /* if num[u] = 0 then */
            T[tCount].u = v; T[tCount].v = u; tCount++;  /* T := T ∪ {uv} */
            father[u] = v;                 /* father[u] := v */
            dfs(u);                        /* рекурсия (в псевдокоде выше опечатка: DFS(v) — иметь в виду DFS(u)) */
        }
        else if (num[u] < num[v] && u != father[v]) { /* else if num[u] < num[v] и u ≠ father[v] */
            B[bCount].u = v; B[bCount].v = u; bCount++;  /* B := B ∪ {uv} */
        }
    }
}

int main(void)
{
    scanf("%d %d", &n, &m);
    for (int e = 0; e < m; e++) {          /* читаем рёбра, строим списки смежности */
        int x, y;
        scanf("%d %d", &x, &y);
        list[x][listSize[x]++] = y;
        list[y][listSize[y]++] = x;        /* граф неориентированный */
    }

    i = 1; tCount = 0; bCount = 0;         /* i := 1;  T := ∅;  B := ∅ */

    for (int v = 1; v <= n; v++)           /* for v ∈ V do num[v] := 0 */
        num[v] = 0;

    for (int v = 1; v <= n; v++)           /* for v ∈ V do */
        if (num[v] == 0) {                 /*    if num[v] = 0 then */
            father[v] = 0;                 /*       father[v] := ∅ */
            dfs(v);                        /*       DFS(v) */
        }

    printf("Древесные рёбра T:\n");        /* вывод множеств T и B */
    for (int k = 0; k < tCount; k++)
        printf("%d-%d\n", T[k].u, T[k].v);
    printf("Обратные рёбра B:\n");
    for (int k = 0; k < bCount; k++)
        printf("%d-%d\n", B[k].u, B[k].v);
    return 0;
}
```

Этот алгоритм применим к произвольному обыкновенному графу $G$. Если $G$ — связный граф, то цикл в строках 15–19 достаточно заменить вызовом процедуры $DFS(v_0)$, применительно к начальной вершине $v_0$. Заметим, что вершина $v$ просмотрена с начала работы процедуры $DFS(v)$; в тот момент, когда процедура $DFS(v)$ закончила работу, вершина $v$ является использованной.

## Теорема 8.1

Пусть $G$ — связный $(n,m)$-граф. Тогда:

1. Поиск в глубину просматривает каждую вершину в точности один раз.
2. Поиск в глубину требует $O(n+m)$ операций.
3. Подграф $(V,T)$ графа $G$ является деревом.

(Доказательство в пособии [1])

Р-м нерекурсивную версию процедуры $DFS(v)$. Рекурсия организуется с помощью стека $S$, элементами которого являются вершины графа. Вершина $v$ является просмотренной, если $num[v] \neq 0$. Вершина $v$ становится использованной с момента, когда $v = top(S)$ ($v$ находится в вершине стека) и все вершины, смежные с $v$, уже просмотрены (в этом случае $v$ удаляется из стека). Вычисление обратных рёбер, связанное с множеством обратных рёбер $B$, здесь опущено.

```
procedure DFS(v);
begin
  num[v] := i; i := i+1;
  S := ∅; S <= v;
  while S ≠ ∅ do
    begin
      v := top(S);
      if exists u, u ∈ list[v] and num[u] = 0
      then
        begin
          num[u] := i; i := i+1; T := T ∪ {uv};
          father[u] := v; S <= u;
        end
      else v <= S
    end
end.
```

Нерекурсивная версия на Си — стек $S$ объявлен явно, как в псевдокоде. Каркас `main` (ввод графа, обнуление `num`, цикл по непомеченным вершинам с вызовом `dfs`) берётся без изменений из примера выше — меняется только сама процедура; обратные рёбра $B$ здесь, как и в псевдокоде, не вычисляются, поэтому в печати остаётся только `T`:

```c
int stack[MAXV];   /* стек S: элементы — вершины графа */
int top = 0;       /* top = 0 — пустой стек (S := ∅) */

/* procedure DFS(v); — нерекурсивная версия */
void dfs(int v)
{
    num[v] = i; i = i + 1;               /* num[v] := i;  i := i+1 */

    top = 0;                             /* S := ∅ */
    stack[top++] = v;                    /* S <= v  (положить v в стек) */

    while (top > 0) {                    /* while S ≠ ∅ do */
        v = stack[top - 1];              /* v := top(S) */

        int exists = 0;                  /* if exists u, u ∈ list[v] и num[u] = 0 */
        for (int k = 0; k < listSize[v]; k++) {
            int u = list[v][k];
            if (num[u] == 0) {           /* непомеченный сосед найден */
                num[u] = i; i = i + 1;   /*    num[u] := i;  i := i+1 */
                T[tCount].u = v; T[tCount].v = u; tCount++;  /* T := T ∪ {uv} */
                father[u] = v;           /*    father[u] := v */
                stack[top++] = u;        /*    S <= u  (push) */
                exists = 1;
                break;                   /*    «выбираем любую» — берём первую */
            }
        }
        if (!exists)
            top--;                       /* else v <= S  (pop): вершина использована */
    }
}
```
